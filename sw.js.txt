/* Webround - cache per l'uso senza rete */
var C='webround-v1';
var CORE=['./','./index.html','./manifest.webmanifest','./icon-192.png','./icon-512.png'];
self.addEventListener('install',function(e){
  e.waitUntil(caches.open(C).then(function(c){return c.addAll(CORE)}).then(function(){return self.skipWaiting()}).catch(function(){}));
});
self.addEventListener('activate',function(e){
  e.waitUntil(caches.keys().then(function(k){
    return Promise.all(k.map(function(n){return n===C?null:caches.delete(n)}));
  }).then(function(){return self.clients.claim()}));
});
self.addEventListener('fetch',function(e){
  if(e.request.method!=='GET') return;
  e.respondWith(
    caches.match(e.request,{ignoreSearch:true}).then(function(hit){
      if(hit) return hit;
      return fetch(e.request).then(function(res){
        var copy=res.clone();
        caches.open(C).then(function(c){c.put(e.request,copy)}).catch(function(){});
        return res;
      }).catch(function(){
        return e.request.mode==='navigate'?caches.match('./index.html'):Response.error();
      });
    })
  );
});
